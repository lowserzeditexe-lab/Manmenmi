foreach(header IN LISTS public_headers)
  string(REPLACE "/" "_" filename "${header}")
  set(source "${CMAKE_CURRENT_BINARY_DIR}/header_${filename}.cpp")
  file(WRITE "${source}" "#include <manmenmi/${header}>\n")
  list(APPEND header_sources "${source}")
endforeach()

function(cli_case name exit_code match forbid)
  add_test(NAME cli.${name} COMMAND "${CMAKE_COMMAND}"
    "-DPROGRAM=$<TARGET_FILE:manmenmi_probe>" "-DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}"
    "-DARGUMENTS=${ARGN}" "-DEXPECT_EXIT=${exit_code}" "-DEXPECT_TEXT=${match}" "-DFORBID_TEXT=${forbid}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/assert_cli.cmake")
  set_tests_properties(cli.${name} PROPERTIES LABELS "cli;foundation" TIMEOUT 30)
endfunction()
